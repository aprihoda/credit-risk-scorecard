98         /*   Woe_OneVariable macro builds the bin table for a single predictor                            */
99         /*   Called by %FineClass_Woe once per variable - not run on its own                              */
100        /*                                                                                                */
101        /**************************************************************************************************/
102        
103        %MACRO Woe_OneVariable(Rank=, Raw=, Out=);
104        
105            %LOCAL TotGood TotBad;
106        
107            PROC MEANS DATA=WORK.Fine NOPRINT NWAY;
108                CLASS &Rank;
109                VAR DefaultFlag &Raw;
110                OUTPUT OUT=SCORE.&Out
111                       n(DefaultFlag)   = NLoans
112                       sum(DefaultFlag) = NBad
113                       min(&Raw)        = RawMin
114                       max(&Raw)        = RawMax;
115            RUN;
116        
117            PROC SQL NOPRINT;
118                SELECT sum(NLoans - NBad), sum(NBad)
119                  INTO :TotGood trimmed, :TotBad trimmed
120                FROM SCORE.&Out;
121            QUIT;
122        
123            DATA SCORE.&Out;
124                SET SCORE.&Out;
2                                                          The SAS System                      Thursday, August  6, 2026 04:06:20 PM

125                NGood   = NLoans - NBad;
126                PctGood = NGood / &TotGood;
127                PctBad  = NBad  / &TotBad;
128                IF NBad > 0 AND NGood > 0 THEN Woe = log(PctGood / PctBad);
129                ELSE Woe = .;
130                IvPart = (PctGood - PctBad) * Woe;
131                FORMAT PctGood PctBad PERCENT8.2 Woe IvPart 8.4;
132                DROP _TYPE_ _FREQ_;
133            RUN;
134        
135            PROC PRINT DATA=SCORE.&Out NOOBS LABEL;
136                TITLE "Fine Bins for &Raw - Weight of Evidence Should Move in One Direction";
137                VAR &Rank RawMin RawMax NLoans NBad Woe IvPart;
138                LABEL &Rank="Bin" RawMin="From" RawMax="To" NLoans="Loans"
139                      NBad="Defaults" Woe="WOE" IvPart="IV contribution";
140            RUN;
141        
142            PROC SQL;
143                TITLE "Information Value for &Raw";
144                SELECT sum(IvPart) AS IV FORMAT=8.4 LABEL="Information Value"
145                FROM SCORE.&Out;
146            QUIT;
147        
148            TITLE;
